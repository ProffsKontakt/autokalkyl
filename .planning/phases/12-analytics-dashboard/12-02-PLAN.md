   ```typescript
   function injectWhereClause(query: string, condition: string): string {
     // If query has WHERE, add AND condition
     // If query has GROUP BY/ORDER BY/LIMIT but no WHERE, add WHERE before them
     // Handle both cases safely
     const upperQuery = query.toUpperCase()

     if (upperQuery.includes('WHERE')) {
       // Find the position after WHERE and its conditions, before GROUP BY/ORDER BY/LIMIT
       const groupByPos = upperQuery.indexOf('GROUP BY')
       const orderByPos = upperQuery.indexOf('ORDER BY')
       const limitPos = upperQuery.indexOf('LIMIT')

       const insertPos = Math.min(
         groupByPos > -1 ? groupByPos : Infinity,
         orderByPos > -1 ? orderByPos : Infinity,
         limitPos > -1 ? limitPos : Infinity,
         query.length
       )

       return query.slice(0, insertPos) + ` AND ${condition} ` + query.slice(insertPos)
     } else {
       // No WHERE clause, find FROM clause and insert WHERE after table reference
       const fromPos = upperQuery.indexOf('FROM')
       if (fromPos === -1) return query // Safety check

       // Find end of FROM clause (next keyword or end)
       const groupByPos = upperQuery.indexOf('GROUP BY')
       const orderByPos = upperQuery.indexOf('ORDER BY')
       const limitPos = upperQuery.indexOf('LIMIT')

       const insertPos = Math.min(
         groupByPos > -1 ? groupByPos : Infinity,
         orderByPos > -1 ? orderByPos : Infinity,
         limitPos > -1 ? limitPos : Infinity,
         query.length
       )

       return query.slice(0, insertPos) + ` WHERE ${condition} ` + query.slice(insertPos)
     }
   }
   ```
