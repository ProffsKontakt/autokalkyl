     trackCalculationCreated,

     trackCalculationDeleted,
     trackWizardCompleted,
   } from '@/lib/analytics/server-events'

2. **Track calculation creation in saveDraft:**
   After successful creation (when `!data.calculationId`):
   ```typescript
   // After: const created = await tenantClient.calculation.create(...)
   if (!data.calculationId) {
     // New calculation created
     await trackCalculationCreated(
       session.user.id,
       created.id,
       effectiveOrgId,
       data.customerName
     )
   } else {
     // Existing calculation updated
     await trackCalculationUpdated(
       session.user.id,
       data.calculationId,
       effectiveOrgId,
       'DRAFT'
     )
   }
   ```

3. **Track status changes in finalizeCalculation:**
   After successful finalization:
   ```typescript
   await trackCalculationUpdated(
     session.user.id,
     calculationId,
     calculation.orgId,
     'FINALIZED'
   )
   // Also track wizard completion
   await trackWizardCompleted(
     session.user.id,
     calculationId,
     calculation.orgId,
     calculation.batteries.length
   )
   ```

4. **Track deletion in deleteCalculation:**
   After successful deletion:
   ```typescript
   await trackCalculationDeleted(
     session.user.id,
     calculationId,
     calculation.orgId
   )
   ```

**Important:**
- Use try/catch around tracking calls to prevent analytics failures from breaking the action
- Log tracking errors but don't fail the action
- Example:
  ```typescript
  try {
    await trackCalculationCreated(...)
  } catch (error) {
    console.error('Analytics tracking failed:', error)
    // Don't fail the action - analytics is non-critical
  }
  ```
