```tsx
'use client';

import { useState, useTransition } from 'react';
import { useRouter } from 'next/navigation';
import { useForm } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { z } from 'zod';
import { toast } from 'sonner';
import { createNatagare } from '@/actions/natagare';

const requestSchema = z.object({
  name: z.string().min(2, 'Namn maaste vara minst 2 tecken').max(100),
  dayRateSekKw: z.number().min(0, 'Dagtariff faar inte vara negativ'),
  nightRateSekKw: z.number().min(0, 'Natttariff faar inte vara negativ'),
  dayStartHour: z.number().int().min(0).max(23),
  dayEndHour: z.number().int().min(0).max(23),
});

type RequestFormData = z.infer<typeof requestSchema>;

export function NatagareRequestForm() {
  const router = useRouter();
  const [isPending, startTransition] = useTransition();

  const { register, handleSubmit, formState: { errors } } = useForm<RequestFormData>({
    resolver: zodResolver(requestSchema),
    defaultValues: {
      dayStartHour: 6,
      dayEndHour: 22,
    }
  });

  const onSubmit = (data: RequestFormData) => {
    startTransition(async () => {
      const result = await createNatagare(data);
      if (result.error) {
        toast.error(result.error);
      } else {
        toast.success('Naetaegare begraerd! Den aer nu synlig foer din organisation.');
        router.push('/dashboard/natagare');
        router.refresh();
      }
    });
  };

  return (
    <form onSubmit={handleSubmit(onSubmit)} className="space-y-6">
      {/* Info banner */}
      <div className="p-4 bg-blue-50 dark:bg-blue-900/20 rounded-lg border border-blue-200 dark:border-blue-800">
        <p className="text-sm text-blue-800 dark:text-blue-200">
          <strong>Hur det fungerar:</strong> Din begraerda naetaegare blir synlig foer din
          organisation direkt och kan anvaendas i kalkyler. Super Admin graanskar och
          godkaenner foer att goera den tillgaenglig foer alla organisationer.
        </p>
      </div>

      {/* Form fields: name, dayRateSekKw, nightRateSekKw, dayStartHour, dayEndHour */}
      {/* Similar to existing natagare-form.tsx but simpler */}

      <button
        type="submit"
        disabled={isPending}
        className="w-full py-2 px-4 bg-blue-600 text-white rounded-md hover:bg-blue-700 disabled:opacity-50"
      >
        {isPending ? 'Skickar begaeran...' : 'Begaer naetaegare'}
      </button>
    </form>
  );
}
```

**Create src/components/dashboard/pending-approvals-widget.tsx:**

Widget for Super Admin dashboard showing pending natagare:

```tsx
import { Card, CardHeader, CardTitle, CardContent } from '@/components/ui/card';
import { Clock, Network } from 'lucide-react';
import { formatDistanceToNow } from 'date-fns';
import { sv } from 'date-fns/locale';
import Link from 'next/link';

interface PendingApprovalsWidgetProps {
  pendingNatagare: Array<{
    id: string;
    name: string;
    requestedByOrgName?: string;
    createdAt: Date;
    approvalStatus: string;
  }>;
}

export function PendingApprovalsWidget({ pendingNatagare }: PendingApprovalsWidgetProps) {
  if (pendingNatagare.length === 0) {
    return (
      <Card>
        <CardHeader>
          <CardTitle className="text-sm font-medium flex items-center gap-2">
            <Network className="w-4 h-4" />
            Vaentande Naetaegare
          </CardTitle>
        </CardHeader>
        <CardContent>
          <p className="text-sm text-gray-500">Inga vaentande godkaennanden</p>
        </CardContent>
      </Card>
    );
  }

  return (
    <Card>
      <CardHeader>
        <CardTitle className="text-sm font-medium flex items-center justify-between">
          <span className="flex items-center gap-2">
            <Network className="w-4 h-4" />
            Vaentande Naetaegare
          </span>
          <span className="inline-flex items-center justify-center w-6 h-6 text-xs font-bold text-white bg-amber-500 rounded-full">
            {pendingNatagare.length}
          </span>
        </CardTitle>
      </CardHeader>
      <CardContent>
        <ul className="space-y-3">
          {pendingNatagare.slice(0, 5).map(n => (
            <li key={n.id} className="flex items-start gap-3 pb-3 border-b last:border-0">
              <Clock className="w-4 h-4 text-gray-400 mt-0.5" />
              <div className="flex-1 min-w-0">
                <p className="font-medium text-sm truncate">{n.name}</p>
                <p className="text-xs text-gray-500">
                  {n.requestedByOrgName || 'Okaaend organisation'} •{' '}
                  {formatDistanceToNow(new Date(n.createdAt), { locale: sv, addSuffix: true })}
                </p>
                {n.approvalStatus === 'DUPLICATE_REVIEW' && (
                  <span className="text-xs text-amber-600">Duplicat - granska</span>
                )}
              </div>
            </li>
          ))}
        </ul>
        {pendingNatagare.length > 5 && (
          <p className="mt-3 text-xs text-gray-500">
            + {pendingNatagare.length - 5} till...
          </p>
        )}
        <Link
          href="/dashboard/admin/natagare"
          className="mt-4 block text-sm text-blue-600 hover:text-blue-700 font-medium"
        >
          Granska alla →
        </Link>
      </CardContent>
    </Card>
  );
}
```
