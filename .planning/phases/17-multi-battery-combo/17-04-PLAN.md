'use client'

import type { UnitBreakdown } from '@/lib/calculations/combo-calculations'

interface ComboBreakdownProps {
  units: UnitBreakdown[]
}

function formatSek(n: number): string {
  return Math.round(n).toLocaleString('sv-SE') + ' kr'
}

export function ComboBreakdown({ units }: ComboBreakdownProps) {
  if (units.length === 0) return null

  return (
    <div className="bg-white border rounded-lg overflow-hidden">
      <div className="px-6 py-4 border-b border-gray-200 bg-gray-50">
        <h3 className="font-medium text-gray-900">Per-enhet breakdown</h3>
        <p className="text-sm text-gray-500">
          Klicka för att expandera varje batteri
        </p>
      </div>

      <div className="divide-y divide-gray-200">
        {units.map((unit, index) => (
          <details key={index} className="group">
            <summary className="px-6 py-4 cursor-pointer hover:bg-gray-50 flex items-center justify-between">
              <div className="flex items-center gap-3">
                <span className="text-sm font-medium text-gray-900">
                  {unit.quantity}x {unit.batteryName}
                </span>
                <span className="text-sm text-gray-500">
                  {unit.capacityKwh} kWh
                </span>
              </div>
              <div className="flex items-center gap-4">
                <span className="text-sm font-medium text-gray-900">
                  {formatSek(unit.subtotalAnnualSavingsSek)}/år
                </span>
                <svg
                  className="w-5 h-5 text-gray-400 group-open:rotate-180 transition-transform"
                  fill="none"
                  stroke="currentColor"
                  viewBox="0 0 24 24"
                >
                  <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M19 9l-7 7-7-7" />
                </svg>
              </div>
            </summary>

            <div className="px-6 pb-6 bg-gray-50">
              <div className="grid grid-cols-2 gap-4 text-sm mt-4">
                {/* Per unit */}
                <div>
                  <h5 className="font-medium text-gray-700 mb-2">Per enhet</h5>
                  <dl className="space-y-1">
                    <div className="flex justify-between">
                      <dt className="text-gray-500">Kapacitet</dt>
                      <dd className="font-medium">{unit.perUnitCapacityKwh} kWh</dd>
                    </div>
                    <div className="flex justify-between">
                      <dt className="text-gray-500">Max urladdning</dt>
                      <dd className="font-medium">{unit.perUnitMaxDischargeKw} kW</dd>
                    </div>
                    <div className="flex justify-between">
                      <dt className="text-gray-500">Kostnad ex. moms</dt>
                      <dd className="font-medium">{formatSek(unit.perUnitCostExVat)}</dd>
                    </div>
                    <div className="flex justify-between">
                      <dt className="text-gray-500">Årlig besparing</dt>
                      <dd className="font-medium">{formatSek(unit.perUnitAnnualSavingsSek)}</dd>
                    </div>
                  </dl>
                </div>

                {/* Subtotal (quantity * per unit) */}
                <div>
                  <h5 className="font-medium text-gray-700 mb-2">
                    Delsumma ({unit.quantity} st)
                  </h5>
                  <dl className="space-y-1">
                    <div className="flex justify-between">
                      <dt className="text-gray-500">Kapacitet</dt>
                      <dd className="font-medium">{unit.subtotalCapacityKwh.toFixed(1)} kWh</dd>
                    </div>
                    <div className="flex justify-between">
                      <dt className="text-gray-500">Max urladdning</dt>
                      <dd className="font-medium">{unit.subtotalMaxDischargeKw.toFixed(1)} kW</dd>
                    </div>
                    <div className="flex justify-between">
                      <dt className="text-gray-500">Kostnad ex. moms</dt>
                      <dd className="font-medium">{formatSek(unit.subtotalCostExVat)}</dd>
                    </div>
                    <div className="flex justify-between">
                      <dt className="text-gray-500">Årlig besparing</dt>
                      <dd className="font-medium">{formatSek(unit.subtotalAnnualSavingsSek)}</dd>
                    </div>
                  </dl>
                </div>
              </div>

              {/* Grid services detail if applicable */}
              {unit.gridServicesIncomeSek > 0 && (
                <div className="mt-4 pt-4 border-t border-gray-200">
                  <div className="flex justify-between text-sm">
                    <span className="text-gray-500">
                      Stödtjänster ({unit.quantity} enheter x {formatSek(unit.perUnitGridServicesIncomeSek)})
                    </span>
                    <span className="font-medium text-purple-600">
                      {formatSek(unit.gridServicesIncomeSek)}
                    </span>
                  </div>
                </div>
              )}
            </div>
          </details>
        ))}
      </div>
    </div>
  )
}
